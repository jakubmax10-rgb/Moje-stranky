        function startNewGame() {
            targetPlayer = playersDB[Math.floor(Math.random() * playersDB.length)];
            guessedPlayers = [];
            guessDetails = [];
            guessesList.innerHTML = '';

            modalOverlay.classList.remove('active');
            
            bestHints = {
                team: { val: "???", status: "" },

            updateTopHintsDisplay();

        input.addEventListener('input', function() {
            renderAutocomplete(this.value);
        });

        input.addEventListener('keydown', function(e) {
            const items = autocompleteList.getElementsByClassName('autocomplete-item');
            if (e.key === 'Enter') {
                e.preventDefault();
                if (items.length > 0) {
                    items[0].click();
                }
            }
        });

        function renderAutocomplete(valQuery) {
            autocompleteList.innerHTML = '';
            const val = valQuery.toLowerCase().trim();
            if (!val) return;

            const matches = playersDB.filter(p => 
                p.name.toLowerCase().includes(val) && 
                !guessedPlayers.includes(p.name)
            );

            matches.forEach(player => {
                const item = document.createElement('div');
                item.className = 'autocomplete-item';
                item.textContent = `${player.name} (${player.team})`;
                item.addEventListener('click', () => {
                    makeGuess(player);
                    input.value = '';
                    autocompleteList.innerHTML = '';
                });
                autocompleteList.appendChild(item);
            });
        }

        function makeGuess(player) {
            guessedPlayers.push(player.name);

            const row = document.createElement('div');
            row.className = 'guess-row';

            // 1. Jméno
            const nameCell = document.createElement('div');
            nameCell.className = 'guess-cell cell-name';
            nameCell.textContent = player.name;
            row.appendChild(nameCell);

            // 2. Tým
            const isTeamCorrect = player.team === targetPlayer.team;
            const teamStatus = isTeamCorrect ? 'correct' : 'incorrect';
            const teamCell = document.createElement('div');
            teamCell.className = `guess-cell ${teamStatus}`;
            teamCell.textContent = player.team;
            row.appendChild(teamCell);
            
            if (isTeamCorrect || bestHints.team.status !== 'correct') {
                bestHints.team = { val: player.team, status: teamStatus };
            }

            // 3. Pozice
            const isPosCorrect = player.pos === targetPlayer.pos;
            const posStatus = isPosCorrect ? 'correct' : 'incorrect';
            const posCell = document.createElement('div');
            posCell.className = `guess-cell ${posStatus}`;
            posCell.textContent = player.pos;
            row.appendChild(posCell);

            if (isPosCorrect || bestHints.pos.status !== 'correct') {
                bestHints.pos = { val: player.pos, status: posStatus };
            }

            // 4. Věk
            const isAgeCorrect = player.age === targetPlayer.age;
            const ageDiff = Math.abs(player.age - targetPlayer.age);
            let ageArrow = player.age < targetPlayer.age ? ' ⬆️' : (player.age > targetPlayer.age ? ' ⬇️' : '');
            
            let ageStatus = 'incorrect';
            if (isAgeCorrect) ageStatus = 'correct';
            else if (ageDiff <= 5) ageStatus = 'partial';

            const ageCell = document.createElement('div');
            ageCell.className = `guess-cell ${ageStatus}`;
            ageCell.textContent = player.age + ageArrow;
            row.appendChild(ageCell);

            if (isAgeCorrect || (bestHints.age.status !== 'correct' && (ageStatus === 'partial' || bestHints.age.status === ''))) {
                bestHints.age = { val: player.age + ageArrow, status: ageStatus };
            }

            // 5. Číslo dresu
            const isNumCorrect = player.number === targetPlayer.number;
            const numDiff = Math.abs(player.number - targetPlayer.number);
            let numArrow = player.number < targetPlayer.number ? ' ⬆️' : (player.number > targetPlayer.number ? ' ⬇️' : '');
            
            let numStatus = 'incorrect';
            if (isNumCorrect) numStatus = 'correct';
            else if (numDiff <= 5) numStatus = 'partial';

            const numCell = document.createElement('div');
            numCell.className = `guess-cell ${numStatus}`;
            numCell.textContent = player.number + numArrow;
            row.appendChild(numCell);

            if (isNumCorrect || (bestHints.number.status !== 'correct' && (numStatus === 'partial' || bestHints.number.status === ''))) {
                bestHints.number = { val: player.number + numArrow, status: numStatus };
            }

            // 6. Národnost
            const isNatCorrect = player.nat === targetPlayer.nat;
            const natStatus = isNatCorrect ? 'correct' : 'incorrect';
            const natCell = document.createElement('div');
            natCell.className = `guess-cell ${natStatus}`;
            natCell.textContent = getFlag(player.nat);
            row.appendChild(natCell);

            if (isNatCorrect || bestHints.nat.status !== 'correct') {
                bestHints.nat = { val: getFlag(player.nat), status: natStatus };
            }

            // Uložit stavy pro Wordle mřížku
            guessDetails.push({
                team: teamStatus,
                pos: posStatus,
                age: ageStatus,
                number: numStatus,
                nat: natStatus
            });

            updateTopHintsDisplay();
            guessesList.insertBefore(row, guessesList.firstChild);

            if (player.name === targetPlayer.name) {
                revealTarget();
                score++;
                streak++;
                saveStats();
                showModal("Skvělá práce! 🏒", `Uhodl jsi hráče <strong>${targetPlayer.name}</strong> na ${guessedPlayers.length} pokus(y).`);
            }
        }

        function revealTarget() {
            document.getElementById('secretCard').textContent = '✔';
            bestHints = {
                team: { val: targetPlayer.team, status: 'correct' },
                pos: { val: targetPlayer.pos, status: 'correct' },
                age: { val: targetPlayer.age, status: 'correct' },
                number: { val: targetPlayer.number, status: 'correct' },
                nat: { val: getFlag(targetPlayer.nat), status: 'correct' }
            };
            updateTopHintsDisplay();
        }

        function saveStats() {
            localStorage.setItem('elh_score', score);
            localStorage.setItem('elh_streak', streak);
            scoreDisplay.textContent = score;
            streakDisplay.textContent = streak;
        }

        function showModal(title, text) {
            modalTitle.textContent = title;
            modalText.innerHTML = text;
            modalOverlay.classList.add('active');
        }

        const hintPickerOverlay = document.getElementById('hintPickerOverlay');
        const hintPickerClose = document.getElementById('hintPickerClose');
        const hintPickerOptions = document.querySelectorAll('.hint-picker-option');

        function fieldDisplayValue(field) {
            if (field === 'nat') return getFlag(targetPlayer.nat);
            return targetPlayer[field];
        }

        function openHintPicker() {
            hintPickerOptions.forEach(opt => {
                const field = opt.dataset.field;
                const alreadyRevealed = bestHints[field].status === 'correct';
                opt.disabled = alreadyRevealed;
                opt.querySelector('.opt-value').textContent = alreadyRevealed ? fieldDisplayValue(field) : '';
            });
            hintPickerOverlay.classList.add('active');
        }

        function closeHintPicker() {
            hintPickerOverlay.classList.remove('active');
        }

        hintBtn.addEventListener('click', openHintPicker);
        hintPickerClose.addEventListener('click', closeHintPicker);
        hintPickerOverlay.addEventListener('click', (e) => {
            if (e.target === hintPickerOverlay) closeHintPicker();
        });

        hintPickerOptions.forEach(opt => {
            opt.addEventListener('click', () => {
                const field = opt.dataset.field;
                if (bestHints[field].status === 'correct') return;
                bestHints[field] = { val: fieldDisplayValue(field), status: 'correct' };
                updateTopHintsDisplay();
                closeHintPicker();
            });
        });

        shareBtn.addEventListener('click', () => {
            const statusToEmoji = (status) => {
                if (status === 'correct') return '🟩';
                if (status === 'partial') return '🟨';
                return '🟥';
            };

            let gridText = guessDetails.map(g => 
                statusToEmoji(g.team) + 
                statusToEmoji(g.pos) + 
                statusToEmoji(g.age) + 
                statusToEmoji(g.number) + 
                statusToEmoji(g.nat)
            ).join('\n');

            const shareText = `ELH Hráč Quiz - Uhodnuto na ${guessedPlayers.length} pokusů! 🏒🇨🇿\n\n${gridText}`;
            
            navigator.clipboard.writeText(shareText).then(() => {
                alert("Výsledek a mřížka zkopírovány do schránky!");
            });
        });
